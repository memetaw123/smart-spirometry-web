Inisialisasi Chart.js const ctx = document.getElementById('liveChart').getContext('2d'); const liveChart = new Chart(ctx, { type: 'line', data: { labels: [], datasets: [{ label: 'Aliran Napas (L/min)', data: [], borderColor: '#2dd4bf', backgroundColor: 'rgba(45, 212, 191, 0.1)', borderWidth: 2, fill: true, tension: 0.3 }] }, options: { responsive: true, maintainAspectRatio: false, scales: { x: { ticks: { color: '#94a3b8' }, grid: { color: '#334155' } }, y: { min: 0, max: 1000, ticks: { color: '#94a3b8' }, grid: { color: '#334155' } } }, plugins: { legend: { labels: { color: '#f8fafc' } } } } }); // 2. Fungsi Update Badge Zona & Angka Laju Aliran function updateZoneBadge(status, pressure) { const badge = document.getElementById('zone-badge'); document.getElementById('pressure-value').innerText = pressure; if (status === 'Zona Merah' || pressure < 500) { badge.className = "w-full text-center py-3 rounded-lg font-bold text-lg bg-red-500/20 text-red-400 border border-red-500/40 shadow-red-500/10"; badge.innerText = "🔴 ZONA MERAH (Rendah)"; } else if (status === 'Zona Kuning' || (pressure >= 500 && pressure < 800)) { badge.className = "w-full text-center py-3 rounded-lg font-bold text-lg bg-yellow-500/20 text-yellow-400 border border-yellow-500/40 shadow-yellow-500/10"; badge.innerText = "🟡 ZONA KUNING (Sedang)"; } else { badge.className = "w-full text-center py-3 rounded-lg font-bold text-lg bg-emerald-500/20 text-emerald-400 border border-emerald-500/40 shadow-emerald-500/10"; badge.innerText = "🟢 ZONA HIJAU (Normal)"; } } // 3. Ambil Riwayat dari Backend API secara Berkala (Polling) async function fetchLatestData() { try { const res = await fetch('/api/history'); const data = await res.json(); const tbody = document.getElementById('history-table-body'); tbody.innerHTML = ''; data.forEach(item => { const row = document.createElement('tr'); const dateFormatted = new Date(item.created_at).toLocaleString('id-ID'); row.innerHTML = ` ${dateFormatted} ${item.device_id} ${item.pressure} L/min ${item.zone_status} `; tbody.appendChild(row); }); // Jika terdapat data terbaru dari database Aiven if (data.length > 0) { const latest = data[0]; updateZoneBadge(latest.zone_status, latest.pressure); // Update Grafik secara Otomatis const timeLabel = new Date(latest.created_at).toLocaleTimeString('id-ID'); if (!liveChart.data.labels.includes(timeLabel)) { if (liveChart.data.labels.length > 15) { liveChart.data.labels.shift(); liveChart.data.datasets[0].data.shift(); } liveChart.data.labels.push(timeLabel); liveChart.data.datasets[0].data.push(latest.pressure); liveChart.update(); } } } catch (err) { console.error("Gagal mengambil data terbaru:", err); } } // Jalankan fetch pertama kali dan ulangi setiap 2 detik (2000 ms) fetchLatestData(); setInterval(fetchLatestData, 2000);
