// Variable timer reset otomatis let autoResetTimer = null; // Function update tampilan dashboard ke kondisi 0 (Standby) function resetDisplayToZero() { document.getElementById('pressure-value').innerText = '0'; const badge = document.getElementById('zone-badge'); badge.className = "w-full text-center py-3 rounded-lg font-bold text-lg bg-slate-700 text-slate-300 transition-all duration-300 shadow-inner"; badge.innerText = "Siap Ditiup (0 L/min)"; } async function fetchLatestData() { try { const res = await fetch('/api/history'); const data = await res.json(); const tbody = document.getElementById('history-table-body'); if (tbody) tbody.innerHTML = ''; data.forEach(item => { const row = document.createElement('tr'); const dateFormatted = new Date(item.created_at).toLocaleString('id-ID'); row.innerHTML = ` ${dateFormatted} ${item.device_id} ${item.pressure} L/min ${item.zone_status} `; tbody.appendChild(row); }); if (data.length > 0) { const latest = data[0]; const dataTime = new Date(latest.created_at).getTime(); const nowTime = new Date().getTime(); // Jika data diterima kurang dari 5 detik lalu, tampilkan hasilnya if (nowTime - dataTime <= 5000) { updateZoneBadge(latest.zone_status, latest.pressure); // Set timer otomatis reset tampilan web ke 0 setelah 5 detik clearTimeout(autoResetTimer); autoResetTimer = setTimeout(() => { resetDisplayToZero(); }, 5000); } else { // Jika sudah lebih dari 5 detik, reset angka utama ke 0 resetDisplayToZero(); } // Update grafik const timeLabel = new Date(latest.created_at).toLocaleTimeString('id-ID'); if (typeof liveChart !== 'undefined' && !liveChart.data.labels.includes(timeLabel)) { if (liveChart.data.labels.length > 15) { liveChart.data.labels.shift(); liveChart.data.datasets[0].data.shift(); } liveChart.data.labels.push(timeLabel); liveChart.data.datasets[0].data.push(latest.pressure); liveChart.update(); } } } catch (err) { console.error("Gagal mengambil data terbaru:", err); } } // Jalankan polling berkala setiap 2 detik fetchLatestData(); setInterval(fetchLatestData, 2000);
